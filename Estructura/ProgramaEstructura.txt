def simulacionZp(p):
    b=[1,0,0,0]
    unos=0
    contador=0
    contador2=0
    lista=list(range(0,p))
    for i in range(0,p):
        for j in range(0,p):
            for k in range(0,p):
                for l in range(0,p):
                    for n in range(0,p):
                        for o in range(0,p):
                            for q in range(0,p):
                                for r in range(0,p):
                                    x=[lista[i],lista[j],lista[k],lista[l]]
                                    y=[lista[n],lista[o],lista[q],lista[r]]

                                    z=[(x[0]*y[0]-x[1]*y[1]-x[2]*y[2]-x[3]*y[3])%p,\
                                        (x[0]*y[1]+x[1]*y[0]+x[2]*y[3]-x[3]*y[2])%p,\
                                        (x[0]*y[2]+x[2]*y[0]+x[3]*y[1]-x[1]*y[3])%p,\
                                        (x[0]*y[3]+x[3]*y[0]+x[1]*y[2]-x[2]*y[1])%p]
                                    contador=contador+1
                                    if z[0]==b[0] and z[1]==b[1] and z[2]==b[2] and z[3]==b[3]:
                                        unos=unos+1

                                    if contador==(p**4):
                                        contador=0
                                        if unos!=0:
                                            unos=0
                                            contador2=contador2+1
                                    print(x,"*",y,":",z)

    print("Dav:",contador2)
    print("p^4:",p**4)
